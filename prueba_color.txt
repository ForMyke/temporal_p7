        var URL = "/Servicio/rest/ws";
        var id_usuario;
        var token;
        var foto = null;

        function get(id) { return document.getElementById(id); }

        var sectionDisplay = {
            login:             'grid',
            alta_usuario:      'grid',
            menu:              'block',
            consulta_usuario:  'grid',

        function muestra(id) { get(id).style.display = sectionDisplay[id] || 'block'; }
        function oculta(id)  { get(id).style.display = "none"; }

        function readSingleFile(files, imagen) {
            var file = files[0];
            if (!file) return;
            var reader = new FileReader();
            reader.onload = function(e) {
                imagen.src = reader.result;
                foto = reader.result.split(',')[1];
            };
            reader.readAsDataURL(file);
        }

        function sha256(ascii) {
            function rightRotate(value, amount) {
                return (value >>> amount) | (value << (32 - amount));
            }
            var mathPow = Math.pow;
            var maxWord = mathPow(2, 32);
            var lengthProperty = 'length';
            var i, j;
            var result = '';
            var words = [];
            var asciiBitLength = ascii[lengthProperty] * 8;
            var hash = sha256.h = sha256.h || [];
            var k = sha256.k = sha256.k || [];
            var primeCounter = k[lengthProperty];
            var isComposite = {};
            for (var candidate = 2; primeCounter < 64; candidate++) {
                if (!isComposite[candidate]) {
                    for (i = 0; i < 313; i += candidate) isComposite[i] = candidate;
                    hash[primeCounter] = (mathPow(candidate, .5) * maxWord) | 0;
                    k[primeCounter++] = (mathPow(candidate, 1 / 3) * maxWord) | 0;
                }
            }
            ascii += '\x80';
            while (ascii[lengthProperty] % 64 - 56) ascii += '\x00';
            for (i = 0; i < ascii[lengthProperty]; i++) {
                j = ascii.charCodeAt(i);
                if (j >> 8) throw new Error('Solo se soporta ASCII');
                words[i >> 2] |= j << ((3 - i) % 4) * 8;
            }
            words[words[lengthProperty]] = ((asciiBitLength / maxWord) | 0);
            words[words[lengthProperty]] = (asciiBitLength);
            for (j = 0; j < words[lengthProperty];) {
                var w = words.slice(j, j += 16);
                var oldHash = hash.slice(0);
                for (i = 0; i < 64; i++) {
                    var w15 = w[i - 15], w2 = w[i - 2], a = hash[0], e = hash[4];
                    var temp1 = hash[7]
                        + (rightRotate(e, 6) ^ rightRotate(e, 11) ^ rightRotate(e, 25))
                        + ((e & hash[5]) ^ ((~e) & hash[6])) + k[i]
                        + (w[i] = (i < 16) ? w[i] : (
                            w[i - 16] + (rightRotate(w15, 7) ^ rightRotate(w15, 18) ^ (w15 >>> 3))
                            + w[i - 7] + (rightRotate(w2, 17) ^ rightRotate(w2, 19) ^ (w2 >>> 10))
                        ) | 0);
                    var temp2 = (rightRotate(a, 2) ^ rightRotate(a, 13) ^ rightRotate(a, 22))
                        + ((a & hash[1]) ^ (a & hash[2]) ^ (hash[1] & hash[2]));
                    hash = [(temp1 + temp2) | 0].concat(hash);
                    hash[4] = (hash[4] + temp1) | 0;
                    hash.pop();
                }
                for (i = 0; i < 8; i++) hash[i] = (hash[i] + oldHash[i]) | 0;
            }
            for (i = 0; i < 8; i++)
                for (j = 3; j + 1; j--) {
                    var b = (hash[i] >> (j * 8)) & 255;
                    result += ((b < 16) ? 0 : '') + b.toString(16);
                }
            return result;
        }

        function limpia_login() {
            get("login_email").value = "";
            get("login_password").value = "";
        }

        function login() {
            var cliente = new WSClient(URL);
            var password = get("login_password").value;
            cliente.get("login", {email: get("login_email").value, password: sha256(password)},
                function(code, result) {
                    if (code == 200) {
                        id_usuario = result.id_usuario;
                        token = result.token;
                        oculta("login");
                        muestra("menu");
                    } else alert(JSON.stringify(result));
                });
        }

        function limpia_alta() {
            get("alta_email").value = "";
            get("alta_password").value = "";
            get("alta_nombre").value = "";
            get("alta_apellido_paterno").value = "";
            get("alta_apellido_materno").value = "";
            get("alta_fecha_nacimiento").value = "";
            get("alta_telefono").value = "";
            get("alta_genero").value = "";
            get("alta_imagen").src = "/usuario_sin_foto.png";
            foto = null;
        }

        function alta() {
            var cliente = new WSClient(URL);
            var password = get("alta_password").value;
            var usuario = {
                email: get("alta_email").value,
                password: password != "" ? sha256(password) : password,
                nombre: get("alta_nombre").value,
                apellido_paterno: get("alta_apellido_paterno").value,
                apellido_materno: get("alta_apellido_materno").value != "" ? get("alta_apellido_materno").value : null,
                fecha_nacimiento: get("alta_fecha_nacimiento").value != "" ? new Date(get("alta_fecha_nacimiento").value).toISOString() : null,
                telefono: get("alta_telefono").value != "" ? get("alta_telefono").value : null,
                genero: get("alta_genero").value == "Masculino" ? "M" : get("alta_genero").value == "Femenino" ? "F" : null,
                foto: foto
            };
            cliente.post("alta_usuario", usuario, function(code, result) {
                if (code == 200) {
                    alert("Se registró el usuario");
                    limpia_login();

        function formatearFecha(fecha) {
            var fecha = new Date(fecha);
            var año  = fecha.getFullYear();
            var mes  = (fecha.getMonth() + 1).toString().padStart(2, '0');
            var dia  = fecha.getDate().toString().padStart(2, '0');
            var horas    = fecha.getHours().toString().padStart(2, '0');
            var minutos  = fecha.getMinutes().toString().padStart(2, '0');
            var segundos = fecha.getSeconds().toString().padStart(2, '0');
            return año + "-" + mes + "-" + dia + "T" + horas + ":" + minutos + ":" + segundos;
        }

        function consulta() {
            var cliente = new WSClient(URL);
            cliente.get("consulta_usuario", {id_usuario: id_usuario, token: token},
                function(code, result) {
                    if (code == 200) {
                        limpia_consulta();
                        oculta('menu');
                        muestra('consulta_usuario');

                        get("consulta_password").value = "";

                        get("consulta_apellido_materno").value = result.apellido_materno != null ? result.apellido_materno : "";

                        get("consulta_telefono").value = result.telefono != null ? result.telefono : "";
                        get("consulta_genero").value = result.genero == "M" ? "Masculino" : result.genero == "F" ? "Femenino" : "";

        function readArticuloFile(files, imagen) {
            var file = files[0];
            if (!file) return;
            var reader = new FileReader();
            reader.onload = function(e) {
                imagen.src = reader.result;
                foto_articulo = reader.result.split(',')[1];
            };
            reader.readAsDataURL(file);
        }

        function limpia_captura_articulo() {
            get("art_nombre").value = "";
            get("art_descripcion").value = "";
            get("art_precio").value = "";
            get("art_cantidad").value = "";
            get("art_imagen").src = "/usuario_sin_foto.png";

        function captura_articulo() {
            var articulo = {
                nombre: get("art_nombre").value,
                descripcion: get("art_descripcion").value != "" ? get("art_descripcion").value : null,
                precio: get("art_precio").value != "" ? parseFloat(get("art_precio").value) : null,
                cantidad: get("art_cantidad").value != "" ? parseInt(get("art_cantidad").value) : null,
                foto: foto_articulo
            };
            fetch(URL + "/alta_articulo?id_usuario=" + id_usuario + "&token=" + encodeURIComponent(token),
                {method: "POST", headers: {"Content-Type": "application/json"}, body: JSON.stringify(articulo)})
                .then(function(r) {
                    return r.json().then(function(data) {
                        if (r.status == 200) {
                            alert("Se dio de alta el artículo");
                            limpia_captura_articulo();
                        } else alert(JSON.stringify(data));
                    });
                })
                .catch(function(e) { alert("Error: " + e.message); });
        }

        function escHtml(s) {
            if (!s) return "";
            return s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
        }

        var accentClasses = ['accent-blue','accent-coral','accent-mint','accent-yellow','accent-lblue'];

        function busca_articulos() {
            var cliente = new WSClient(URL);
            cliente.get("consulta_articulos", {
                    palabra: get("busqueda_palabra").value,
                    id_usuario: id_usuario,
                    token: token
                },
                function(code, result) {
                    if (code == 200) {
                        resultados = result;
                        var html = "";
                        for (var i = 0; i < result.length; i++) {
                            var a = result[i];
                            var accent = accentClasses[i % accentClasses.length];
                            var fotoSrc = a.foto ? "data:image/jpeg;base64," + a.foto : "/usuario_sin_foto.png";
                            html += "<div class='art-card-cb " + accent + "'>";
                            html += "<img src='" + fotoSrc + "' alt=''>";
                            html += "<div class='art-card-body'>";
                            html += "<div class='art-card-name'>" + escHtml(a.nombre) + "</div>";
                            html += "<div class='art-card-desc'>" + escHtml(a.descripcion) + "</div>";
                            html += "<div class='art-card-price'>$" + (a.precio != null ? a.precio.toFixed(2) : "0.00") + "</div>";
                            html += "<div class='art-card-footer'>";
                            html += "<input type='number' id='cant_" + i + "' value='1' min='1' class='art-qty-cb'>";
                            html += "<button class='btn-buy-cb' onclick='compra(" + i + ")'><i class='bi bi-cart-plus'></i> Comprar</button>";
                            html += "</div></div></div>";
                        }
                        get("resultados_articulos").innerHTML = html || "<p style='color:#888;font-size:0.85rem;padding:1rem 0'>No se encontraron artículos.</p>";
                    } else alert(JSON.stringify(result));
                });
        }

        function compra(idx) {
            var a = resultados[idx];
            var cantidad = parseInt(get("cant_" + idx).value);
            if (isNaN(cantidad) || cantidad < 1) { alert("La cantidad debe ser mayor a 0"); return; }
            var cliente = new WSClient(URL);
            cliente.put("compra_articulo", {
                    id_articulo: a.id_articulo,
                    cantidad: cantidad,
                    id_usuario: id_usuario,
                    token: token
                }, {},
                function(code, result) {
                    if (code == 200) {
                        var encontrado = false;
                        for (var i = 0; i < carrito.length; i++) {
                            if (carrito[i].id_articulo == a.id_articulo) {
                                carrito[i].cantidad += cantidad;
                                encontrado = true;
                                break;
                            }
                        }
                        if (!encontrado) carrito.push({
                            id_articulo: a.id_articulo,
                            nombre: a.nombre,
                            descripcion: a.descripcion,
                            precio: a.precio,
                            cantidad: cantidad,
                            foto: a.foto
                        });
                        alert("Artículo agregado al carrito");
                    } else alert(JSON.stringify(result));
                });
        }

        function renderiza_carrito() {
            var html = "";
            var total = 0;
            for (var i = 0; i < carrito.length; i++) {
                var a = carrito[i];
                var costo = a.precio * a.cantidad;
                total += costo;
                var fotoSrc = a.foto ? "data:image/jpeg;base64," + a.foto : "/usuario_sin_foto.png";
                html += "<div class='cart-item-row'>";
                html += "<img src='" + fotoSrc + "' alt=''>";
                html += "<div><div class='cart-item-name'>" + escHtml(a.nombre) + "</div>";
                html += "<div class='cart-item-meta'>Cant: " + a.cantidad + " &nbsp;&middot;&nbsp; $" + a.precio.toFixed(2) + " c/u</div>";
                html += "<div class='cart-item-subtotal'>$" + costo.toFixed(2) + "</div>";
                html += "<button class='btn-remove-cb' onclick='elimina_del_carrito(" + a.id_articulo + ")'><i class='bi bi-trash3'></i> Eliminar</button></div>";
                html += "<div></div></div>";
            }
            if (carrito.length == 0) html = "<p style='color:#888;font-size:0.85rem;font-style:italic'>El carrito está vacío.</p>";
            get("lista_carrito").innerHTML = html;
            get("cart-total-display").textContent = "$" + total.toFixed(2);
        }

        function ir_a_carrito() {
            renderiza_carrito();
            oculta("compra_articulos");
            muestra("articulos_carrito");
        }

        function elimina_del_carrito(id_articulo) {
            var cliente = new WSClient(URL);
            cliente.delete("elimina_articulo_carrito_compra", {
                    id_usuario: id_usuario,
                    id_articulo: id_articulo,
                    token: token
                },
                function(code, result) {
                    if (code == 200) {
                        carrito = carrito.filter(function(a) { return a.id_articulo != id_articulo; });
                        renderiza_carrito();
                    } else alert(JSON.stringify(result));
                });
        }

        function elimina_todo_carrito() {
            if (!confirm("¿Deseas eliminar todos los artículos del carrito?")) return;
            var cliente = new WSClient(URL);
            cliente.delete("elimina_carrito_compra", {id_usuario: id_usuario, token: token},
                function(code, result) {
                    if (code == 200) { carrito = []; renderiza_carrito(); }
                    else alert(JSON.stringify(result));
                });
        }
